#ifndef SORT_H
#define SORT_H

// Bubble Sort (Aula 5)
// Compara pares adjacentes e movimenta o maior valor para o final.
// Encerra antecipadamente se nao houver trocas na passagem (melhor caso O(N)).
void bubbleSort(int *V, int N)
{
    int i, continua, aux, fim = N;
    do
    {
        continua = 0;
        for (i = 0; i < fim - 1; i++)
        {
            if (V[i] > V[i + 1])
            {
                aux = V[i];
                V[i] = V[i + 1];
                V[i + 1] = aux;
                continua = i;
            }
        }
        fim--;
    } while (continua != 0);
}

// Selection Sort (Aula 5)
// A cada passo, procura o menor elemento da parte nao ordenada e troca com a posicao atual.
void selectionSort(int *V, int N)
{
    int i, j, menor, troca;
    for (i = 0; i < N - 1; i++)
    {
        menor = i;
        for (j = i + 1; j < N; j++)
        {
            if (V[j] < V[menor])
            {
                menor = j;
            }
        }
        if (i != menor)
        {
            troca = V[i];
            V[i] = V[menor];
            V[menor] = troca;
        }
    }
}

// Insertion Sort (Aula 7)
// Percorre o vetor deslocando elementos maiores a esquerda e insere o valor na posicao correta.
void insertionSort(int *V, int N)
{
    int i, j, aux;
    for (i = 1; i < N; i++)
    {
        aux = V[i];
        for (j = i; (j > 0) && (aux < V[j - 1]); j--)
        {
            V[j] = V[j - 1];
        }
        V[j] = aux;
    }
}

// Cria o vetor dinamicamente e preenche com inteiros aleatorios de 0 a 1.000.000
int *criarVetores(int numeroElementos)
{
    int i, parte1, parte2;
    int *vetor = (int *)malloc(numeroElementos * sizeof(int));

    if (vetor == NULL)
    {
        printf("Erro ao alocar memoria!\n");
        return NULL;
    }

    for (i = 0; i < numeroElementos; i++)
    {
        // Combina duas chamadas de rand() usando apenas int para superar o limite 32767 no Windows
        parte1 = rand() % 10000;
        parte2 = rand() % 10000;
        vetor[i] = (parte1 * 10000 + parte2) % 1000001;
    }

    return vetor;
}

// Copia o vetor original para preservar o backup e mede o tempo de ordenacao em ms.
// Para vetores pequenos (100 e 1.000), cria copias antes do clock() e divide o tempo total
// pela quantidade de execucoes para evitar leituras zeradas na resolucao de 1 ms do clock().
double medirTempoOrdenacao(int *vetorOriginal, int tamanho, int algoritmo)
{
    int lotes = 1;
    int i, k;
    int *copias;
    clock_t inicio, fim;
    double tempoMs;

    if (tamanho == 100)
    {
        lotes = LOTES_100;
    }
    else if (tamanho == 1000)
    {
        lotes = LOTES_1000;
    }

    copias = (int *)malloc(lotes * tamanho * sizeof(int));
    if (copias == NULL)
    {
        printf("Erro ao alocar memoria para copia do vetor!\n");
        return -1.0;
    }

    // Restaura o backup em todas as copias antes de iniciar o cronometro
    for (k = 0; k < lotes; k++)
    {
        for (i = 0; i < tamanho; i++)
        {
            copias[k * tamanho + i] = vetorOriginal[i];
        }
    }

    // Mede apenas a chamada da funcao de ordenacao ate o seu termino
    inicio = clock();
    if (algoritmo == 1)
    {
        for (k = 0; k < lotes; k++)
        {
            bubbleSort(&copias[k * tamanho], tamanho);
        }
    }
    else if (algoritmo == 2)
    {
        for (k = 0; k < lotes; k++)
        {
            selectionSort(&copias[k * tamanho], tamanho);
        }
    }
    else if (algoritmo == 3)
    {
        for (k = 0; k < lotes; k++)
        {
            insertionSort(&copias[k * tamanho], tamanho);
        }
    }
    fim = clock();

    tempoMs = (((double)(fim - inicio) / CLOCKS_PER_SEC) * 1000.0) / lotes;

    free(copias);

    return tempoMs;
}

1. Introdução

A organização e a manipulação de dados são atividades fundamentais na área de desenvolvimento de sistemas. Entre essas atividades, os algoritmos de ordenação possuem grande importância, pois permitem organizar informações de maneira crescente ou decrescente, facilitando sua consulta e utilização.

A atividade tem como objetivo implementar e analisar três algoritmos de ordenação: Bubble Sort, Selection Sort e Insertion Sort.

A proposta consiste em desenvolver um programa capaz de gerar vetores de números inteiros em diferentes condições de organização e tamanhos, permitindo executar os três algoritmos e comparar seus respectivos tempos de execução, avaliando o desempenho entre eles.

Para a realização dos testes, foram estabelecidas três condições de vetores: aleatórios, crescentes e decrescentes. Também foram definidos cinco tamanhos diferentes, sendo eles 100, 1.000, 10.000, 100.000 e 1.000.000 de elementos.

3. Implementação

O programa foi desenvolvido utilizando a linguagem C, com uma estrutura organizada em dois arquivos principais: main.c e sort.h. Essa divisão foi adotada para separar o controle geral do programa das implementações dos algoritmos de ordenação, facilitando a organização e a manutenção do código.

3.1. Organização dos arquivos

  • main.c: responsável pelo funcionamento principal do programa. Contém os menus de interação com o usuário, a geração dos vetores, a criação de cópias de backup, o gerenciamento de memória, a chamada dos algoritmos e a medição do tempo de execução com a biblioteca <time.h> (clock() e CLOCKS_PER_SEC).
  • sort.h: biblioteca responsável exclusivamente por armazenar as implementações dos três algoritmos de ordenação (bubbleSort, selectionSort e insertionSort), protegida por guardas de inclusão (#ifndef SORT_H).

A integração entre os arquivos ocorre por meio da diretiva #include "sort.h", utilizada no arquivo principal.

3.2. Estruturas de dados utilizadas

A principal estrutura de dados utilizada no projeto é o vetor (int *), responsável por armazenar os números inteiros. Os vetores foram escolhidos por permitirem acesso direto aos elementos por meio de seus índices, facilitando a implementação dos algoritmos de ordenação.

Para possibilitar a utilização de diferentes quantidades de elementos, foi utilizada a alocação dinâmica de memória por meio da função malloc(). Também foi utilizado um ponteiro de ponteiros (int **) na função gerarVariosVetor() para gerenciar os cinco vetores simultaneamente quando o usuário seleciona a opção de todos os tamanhos. Ao final, a função liberarVetores() chama free() para liberar toda a memória alocada e evitar vazamentos.

3.3. Geração dos vetores

  • Vetor aleatório (criarVetores()): os elementos são gerados combinando duas chamadas de rand() ((parte1 * 10000 + parte2) % 1000001) para superar o limite de 32767 do RAND_MAX no Windows, produzindo inteiros de 0 a 1.000.000. A função srand(time(NULL)) inicializa a semente aleatória no início da execução.
  • Vetor crescente (vetorCrescente()): preenchido em ordem crescente, começando pelo número 1 até N.
  • Vetor decrescente (vetorDecrescente()): preenchido em ordem decrescente, começando pela quantidade total de elementos (N) e diminuindo até chegar a 1.

Essa variedade permite observar o comportamento dos algoritmos diante de diferentes condições de organização dos dados.

3.4. Quantidade de elementos

Para atender à proposta do trabalho, foram implementadas cinco opções de tamanho na função seletorValorQuantidadeElementos() (além da opção 6 para gerar todos em conjunto via gerarVariosVetor()):

  • 100 elementos
  • 1.000 elementos
  • 10.000 elementos
  • 100.000 elementos
  • 1.000.000 de elementos

3.5. Cópia e preservação dos vetores

Realizada dentro da função medirTempoOrdenacao(), que aloca o vetor auxiliar copias e copia os dados de vetorOriginal antes de iniciar o cronômetro.

Isso garante que cada algoritmo (e cada repetição dos modos 3x e 5x) receba exatamente a mesma sequência original não modificada, mantendo o backup intacto e assegurando uma comparação justa. Para 100 (LOTES_100 = 10000) e 1.000 (LOTES_1000 = 200) elementos, são preparadas múltiplas cópias em lote antes de medir.

3.6. Algoritmos de ordenação

3.6.1. Bubble Sort Compara elementos vizinhos e realiza trocas quando estão fora de ordem, levando o maior valor restante para o final. A implementação usa um laço do-while com a variável de controle continua, encerrando antecipadamente caso nenhuma troca ocorra na passagem (melhor caso O(n)).
3.6.2. Selection Sort Procura o menor elemento da parte ainda não ordenada do vetor em cada passagem, armazenando seu índice na variável menor. Ao final da varredura interna, realiza a troca com a posição atual apenas se i != menor.
3.6.3. Insertion Sort Organiza os elementos de forma semelhante à ordenação de cartas na mão: armazena o elemento atual em aux, desloca para a direita os elementos anteriores maiores que ele e insere aux na posição correta, exigindo menos deslocamentos quando o vetor já está organizado.

3.7. Medição do tempo de execução

A medição é feita em main.c na função medirTempoOrdenacao() usando <time.h> e clock(). O instante inicial é registrado imediatamente antes da chamada do algoritmo e o instante final logo após sua conclusão, convertendo o resultado para milissegundos com quatro casas decimais (%.4f ms): 

Tempo (ms) = (((double)(fim - inicio) / CLOCKS_PER_SEC) * 1000.0) / lotes

A geração e a cópia dos vetores ocorrem antes de clock() e não entram no tempo medido. Para evitar leituras zeradas pela resolução de 1 ms do clock() no Windows, vetores de 100 e 1.000 elementos são executados em lotes (10.000 e 200 cópias, respectivamente) e o tempo total é dividido por lotes. Quando a opção de todos os tamanhos é escolhida, o programa mede e exibe o tempo individual para cada tamanho — e nas opções 5 (3x) e 6 (5x), calcula e apresenta a média final por tamanho e algoritmo.

3.8. Funcionamento geral do programa

O programa opera por menus interativos em cascata, permitindo escolher o tipo de vetor, o tamanho (individual ou todos os cinco) e o modo de execução: algoritmo único, todos os três métodos uma vez, ou execução em série de 3x ou 5x com exibição automática da média ao final.

4. Estudo de Complexidade (Notação Big-O)

A análise de complexidade permite compreender como o desempenho de um algoritmo varia de acordo com a quantidade de elementos processados. Os três algoritmos implementados apresentam comportamentos distintos, principalmente quando aplicados a vetores já ordenados ou em ordem inversa.

4.1. Complexidade do Bubble Sort

Compara elementos vizinhos e realiza trocas quando necessário. Como a implementação em sort.h utiliza a flag continua para encerrar antecipadamente quando não há trocas em uma passagem, um vetor já ordenado percorre o laço apenas uma vez (O(n)). Nos cenários aleatório e decrescente, o crescimento é quadrático.

  • Melhor caso: O(n)
  • Caso médio: O(n²)
  • Pior caso: O(n²)

4.2. Complexidade do Selection Sort

Procura o menor elemento da parte não ordenada em cada passagem. Mesmo quando os elementos já estão organizados em ordem crescente, o algoritmo percorre toda a parte restante para confirmar o menor valor, mantendo o custo quadrático em todos os cenários.

  • Melhor caso: O(n²)
  • Caso médio: O(n²)
  • Pior caso: O(n²)

4.3. Complexidade do Insertion Sort

Quando o vetor já está ordenado, a condição aux < V[j - 1] falha imediatamente a cada iteração, resultando em comportamento linear. Quando está aleatório ou decrescente, diversos deslocamentos são necessários até posicionar cada valor.

  • Melhor caso: O(n)
  • Caso médio: O(n²)
  • Pior caso: O(n²)

4.4. Comparação das complexidades

Algoritmo Melhor Caso (Crescente) Caso Médio (Aleatório) Pior Caso (Decrescente) Memória Extra
Selection Sort O(n²) O(n²) O(n²) O(1) — In-place
Insertion Sort O(n) O(n²) O(n²) O(1) — In-place
Bubble Sort O(n) O(n²) O(n²) O(1) — In-place

A partir dessa comparação, observa-se que o Insertion Sort e o Bubble Sort otimizado possuem grande vantagem em vetores já ordenados (O(n)), enquanto o Selection Sort mantém complexidade quadrática mesmo nessa condição. Essa expectativa teórica se confirma diretamente nos resultados práticos obtidos.

6. Análise dos Resultados dos Testes

Os testes foram realizados utilizando três tipos de vetores: crescente, decrescente e aleatório, considerando cinco tamanhos diferentes: 100, 1.000, 10.000, 100.000 e 1.000.000 de elementos.

Para cada cenário, foram realizadas cinco execuções dos algoritmos Bubble Sort, Selection Sort e Insertion Sort. Ao final, foi calculada a média dos tempos obtidos, permitindo comparar o comportamento dos métodos de ordenação.

6.1. Vetor crescente

Nos testes com vetores crescentes, o Bubble Sort e o Insertion Sort apresentaram os menores tempos de execução. Esse comportamento está relacionado ao fato de que os elementos já estavam ordenados.

O Bubble Sort otimizado consegue identificar que não ocorreram trocas e encerrar antecipadamente sua execução. O Insertion Sort também apresenta bom desempenho nesse cenário, pois os elementos não precisam ser deslocados.

Já o Selection Sort apresentou crescimento significativo no tempo de execução conforme o tamanho do vetor aumentou. Isso ocorre porque o algoritmo continua procurando o menor elemento nas posições restantes, mesmo quando o vetor já está ordenado.

6.2. Vetor decrescente

Nos testes com vetores decrescentes, os três algoritmos apresentaram aumento considerável no tempo de execução.

O Selection Sort obteve o melhor desempenho nos tamanhos maiores, seguido pelo Insertion Sort. O Bubble Sort apresentou os maiores tempos, chegando a uma média de 887.034,4000 ms para um milhão de elementos.

Esse comportamento ocorre porque o Bubble Sort e o Insertion Sort precisam realizar um grande número de operações para reorganizar os elementos que estão em ordem inversa. O Selection Sort também apresenta complexidade quadrática, mas seu número de trocas é menor, o que contribuiu para seu desempenho nesse cenário.

6.3. Vetor aleatório

Nos testes com vetores aleatórios, o Insertion Sort apresentou os melhores resultados em todos os tamanhos avaliados, seguido pelo Selection Sort. O Bubble Sort apresentou os maiores tempos de execução.

Para um milhão de elementos, por exemplo, o Insertion Sort apresentou média de 226.470,0000 ms, enquanto o Selection Sort registrou 345.243,4000 ms e o Bubble Sort 1.569.710,0000 ms.

Esses resultados demonstram que, embora os três algoritmos apresentem complexidade quadrática no caso médio ou pior caso, existem diferenças práticas de desempenho relacionadas à quantidade de comparações, deslocamentos e trocas realizadas.

6.4. Observação sobre a precisão das medições

Em alguns testes com vetores crescentes, foram registrados valores iguais a 0,0000 ms ou muito próximos de zero. Isso não significa que os algoritmos não consumiram tempo de processamento, mas que a duração da execução foi muito pequena para ser representada com precisão pelo mecanismo de medição utilizado.

Por esse motivo, os resultados devem ser interpretados considerando essa limitação, principalmente nos vetores menores e nos cenários em que os algoritmos executaram em seu melhor caso.

7. Conclusão

O desenvolvimento deste trabalho possibilitou aplicar, na prática, os conhecimentos adquiridos na disciplina de Estrutura de Dados I — especialmente em relação à manipulação de vetores, organização de informações e análise de complexidade — trabalhando conceitos essenciais da linguagem C, como ponteiros (int * e int **), alocação dinâmica de memória (malloc e free), modularização com arquivo de cabeçalho (sort.h), geração de números aleatórios e medição de tempo com clock(). Além disso, o uso de cópias preservadas dos vetores originais garantiu que os três algoritmos fossem avaliados exatamente sob as mesmas condições.

A partir dos testes realizados, foi possível comprovar na prática que o desempenho dos algoritmos de ordenação varia significativamente conforme a quantidade de elementos e a organização inicial dos dados, confirmando o estudo teórico em notação Big-O.

O Bubble Sort apresentou os maiores tempos de execução nos cenários aleatório e decrescente, principalmente para os vetores maiores. Apesar de possuir melhor caso linear (O(n)) quando otimizado para encerrar sem trocas no vetor crescente, seu desempenho é bastante prejudicado quando precisa realizar um volume elevado de trocas.

O Selection Sort apresentou desempenho inferior ao Insertion Sort nos testes com vetores aleatórios e não se beneficia de vetores já ordenados (mantendo complexidade quadrática O(n²) em todos os cenários), mas obteve os melhores resultados nos vetores decrescentes de maior tamanho graças à sua quantidade reduzida de trocas (no máximo uma por passagem).

O Insertion Sort apresentou os melhores resultados nos testes com vetores aleatórios e também teve excelente desempenho nos vetores crescentes (O(n)), nos quais os elementos já estavam ordenados e não exigiam deslocamentos. Entretanto, nos vetores decrescentes, o aumento expressivo de deslocamentos elevou significativamente seu tempo de execução.

Portanto, os resultados demonstram que não existe um único algoritmo que apresente o melhor desempenho em todas as situações: a escolha ideal depende diretamente das características e da organização inicial dos dados. Assim, o trabalho reforça na prática a importância da análise de complexidade na construção de programas mais eficientes.
